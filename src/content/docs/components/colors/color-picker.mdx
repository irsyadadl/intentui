## Anatomy

Compose `ColorPicker` with a custom button and `Popover`. Wrap the button and popover in `PopoverTrigger` so open state and trigger wiring are handled for you. Keep the selected color state on `ColorPicker` separately from the popover.

```tsx
"use client"

import { parseColor } from "@react-stately/color"
import { useState } from "react"
import { Button } from "@/components/ui/button"
import { ColorArea } from "@/components/ui/color-area"
import { ColorField } from "@/components/ui/color-field"
import { ColorPicker } from "@/components/ui/color-picker"
import { ColorSlider, ColorSliderTrack } from "@/components/ui/color-slider"
import { ColorSwatch } from "@/components/ui/color-swatch"
import { ColorThumb } from "@/components/ui/color-thumb"
import { Input } from "@/components/ui/input"
import { Popover, PopoverBody, PopoverTrigger } from "@/components/ui/popover"

export default function ColorPickerExample() {
  const [color, setColor] = useState(parseColor("#0d6efd"))
  return (
    <ColorPicker value={color} onChange={setColor} defaultValue="rgb(120,140,200)">
      <PopoverTrigger>
        <Button intent="plain" data-slot="control">
          <ColorSwatch />
          Select color
        </Button>
        <Popover aria-label="Choose color" className="[--gutter:--spacing(1)]">
          <PopoverBody>
            <div className="space-y-(--gutter)">
              <ColorArea
                colorSpace="rgb"
                defaultValue="rgb(120,140,200)"
                xChannel="red"
                yChannel="green"
                xName="red"
                yName="green"
              />
              <ColorSlider colorSpace="hsb" channel="hue">
                <ColorSliderTrack>
                  <ColorThumb />
                </ColorSliderTrack>
              </ColorSlider>
              <ColorField aria-label="Color">
                <Input />
              </ColorField>
            </div>
          </PopoverBody>
        </Popover>
      </PopoverTrigger>
    </ColorPicker>
  )
}
```

See the [Popover anatomy](/docs/components/overlays/popover#anatomy) for `PopoverTrigger` and controlled open state.
