## Anatomy
```
import { Button } from "@/components/ui/button"
import { Dialog, DialogBody, DialogClose, DialogDescription, DialogFooter, DialogHeader, DialogTitle, DialogTrigger } from "@/components/ui/dialog"
```

```
<DialogTrigger>
  <Button>Open Dialog</Button>
  <Dialog>
    <DialogHeader>
      <DialogTitle>Dialog Title</DialogTitle>
      <DialogDescription>Dialog Description</DialogDescription>
    </DialogHeader>
    <DialogBody>Dialog Body</DialogBody>
    <DialogFooter>
      <DialogClose>Close</DialogClose>
      <Button>Confirm</Button>
    </DialogFooter>
  </Dialog>
</DialogTrigger>
```

## Examples

### Alert dialog
Use `role="alertdialog"` for a critical message that requires an explicit response.
<How toUse="overlays/dialog/alert-dialog-example" />
Alert dialogs cannot be dismissed by clicking outside by default, and the close icon is hidden. Provide an explicit action such as Cancel.

```tsx
<Dialog role="alertdialog" />
```

### Sizes
The dialog defaults to `md`. Available sizes are `2xs`, `xs`, `sm`, `md`, `lg`, `xl`, `2xl`, `3xl`, `4xl`, `5xl`, and `fullscreen`.
<How toUse="overlays/dialog/dialog-size-example" />

### Dialog overlay styles

The overlay uses a subtle 1px blur by default. You can increase it or disable it entirely.

```tsx
<DialogTrigger>
  <Button intent="outline">Rename</Button>
  <Dialog overlay={{ className: "backdrop-blur-2xl" }}>
```

If you want to turn it off, use `backdrop-blur-none`. If you do not need any blur at all, the recommended approach is to remove the blur class from the component defaults.

```diff
-"bg-background/15 backdrop-blur-[1px] motion-reduce:backdrop-blur-none",
+"bg-background/15",
```

Or, you can override it per instance from the component usage:

```tsx
<DialogTrigger>
  <Button intent="outline">Rename</Button>
  <Dialog overlay={{ className: "backdrop-blur-none" }}>
```

### Header
This setup’s super flexible. If you skip adding `DialogTitle` and just drop a string as its child, it'll auto-render as the title. Like this:
```
<DialogHeader>
  Title
</DialogHeader>
```
Wanna customize more? Throw in props like `title` and `description` for a tailored header:
```
<DialogHeader title='Title' description='Description' />
```

### Triggered by menu
You can also trigger the dialog by clicking on a menu item.
<How toUse="overlays/dialog/dialog-triggered-by-menu-example" />
It might be a good idea to extract the dialog into a separate component for better organization.

```
interface DialogActionProps {
  state: string | null
  onOpenChange: () => void
  actionType: { description: string; action: () => void; confirmText: string; title: string }
  disabled: boolean
}

const DialogAction = (props: DialogActionProps) => (
  <Dialog isOpen={props.state !== null} onOpenChange={props.onOpenChange}>
    <DialogHeader>
      <DialogTitle>{props.actionType?.title}</DialogTitle>
      <DialogDescription>{props.actionType?.description}</DialogDescription>
    </DialogHeader>
    <DialogFooter>
      <DialogClose>Cancel</DialogClose>
      <Button
        intent={props.state === "ban" ? "danger" : "primary"}
        className="min-w-24"
        isDisabled={props.disabled}
        onPress={props.actionType?.action}
      >
        {props.disabled ? <Loader variant="spin" /> : props.actionType?.confirmText}
      </Button>
    </DialogFooter>
  </Dialog>
)
```

Then you can use it like this.

```
<DialogAction
  state={state}
  onOpenChange={closeDialog}
  actionType={actionType(state)}
  disabled={loading}
/>
```

With that, now we can modify the `actionType` function to return the initial state.

```tsx
const actionType = (t: string | null) => {
  const initialsState = {
    title: '',
    description: '',
    confirmText: '',
    action: () => {}
  }

  switch (t) {
    case 'delete': ...
    case 'ban': ...
    case 'restore': ...
    default:
      return initialsState
  }
}
```
