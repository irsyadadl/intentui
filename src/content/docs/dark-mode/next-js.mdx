## Next themes
Use `next-themes` to manage dark mode in Next.js. The switcher uses `SunIcon` and `MoonIcon` from `@heroicons/react`.
<PackageCommand command="npm i next-themes @heroicons/react"/>

## Provider and switcher
Create the theme provider and switcher:
<CodeBlock
    source={{
        'theme-provider.tsx': 'dark-mode/nextjs/theme-provider.tsx',
        'theme-switcher.tsx': 'dark-mode/nextjs/theme-switcher.tsx',
    }} />
## Usage
Add the provider to your root layout:

```tsx {1,7-9}
import { Providers } from '@/components/providers'

export default function RootLayout({ children }: Readonly<{ children: React.ReactNode }>) {
  return (
    <html lang="en" suppressHydrationWarning>
      <body>
        <Providers>
          {children}
        </Providers>
      </body>
    </html>
  )
}
```
